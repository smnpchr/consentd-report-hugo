{{/* Renders a list of enum values as comma-separated, i18n-translated string. If list is nil/empty, returns an em-dash. Params: list (slice), prefix (string for i18n key, e.g. "aggregated_") */}} {{ $list := .list }} {{ $prefix := .prefix }} {{ if $list }} {{- $translated := slice -}} {{- range $list -}} {{- $translated = $translated | append (i18n (printf "%s%s" $prefix .)) -}} {{- end -}} {{- delimit $translated ", " -}} {{ else }}—{{ end }}
